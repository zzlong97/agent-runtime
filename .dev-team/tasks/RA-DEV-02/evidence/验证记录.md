## 既有循环导入的只读 HEAD 对照

不是仅更换收集顺序后推定归属。使用下列内存导入器，把所有 `agent_runtime` 模块
从同一 HEAD 的 `git show` 读出执行；不写文件、不检出或修改工作区：

```python
import importlib.abc
import importlib.util
import pathlib
import subprocess
import sys
import traceback

class HeadLoader(importlib.abc.Loader):
    def __init__(self, path): self.path = path
    def create_module(self, spec): return None
    def exec_module(self, module):
        result = subprocess.run(['git', 'show', 'HEAD:' + self.path], capture_output=True, check=True)
        exec(compile(result.stdout.decode('utf-8'), '<HEAD:' + self.path + '>', 'exec'), module.__dict__)

class HeadFinder(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname, path=None, target=None):
        if not (fullname == 'agent_runtime' or fullname.startswith('agent_runtime.')): return None
        stem = pathlib.Path('src', *fullname.split('.'))
        package = (stem / '__init__.py').exists()
        source = stem / '__init__.py' if package else stem.with_suffix('.py')
        if not source.exists(): return None
        return importlib.util.spec_from_loader(fullname, HeadLoader(source.as_posix()), is_package=package)

sys.meta_path.insert(0, HeadFinder())
from agent_runtime.core.config import Settings
try:
    import agent_runtime.runtime.event_repository
except Exception:
    traceback.print_exc()
    raise SystemExit(1)
```
